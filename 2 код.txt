public class License
{
    public string date;
    public string licenseNumber;

    public bool isExpired() { }
}

public class Specialization
{
    public string profession;
}

public class Service
{
    public string name;
    public float price;
}

public abstract class BaseDoctor
{
    public string name;
    public License license;
    public Specialization[] specializations;
    public Service[] services;

    public abstract bool checkService(Service service);
    public bool checkLicense();
    public abstract string[] getSchedule();
}

public abstract class InvitedDoctor : BaseDoctor
{
    public bool checkService(Service serive) {}
    public string[] getSchedule() {}
}

public abstract class RegularDoctor : BaseDoctor
{
    public bool checkService(Service serive) {}
    public string[] getSchedule() {}
}

public class Patient
{
    public string name;
    public string phone;
    public string email;
}

public class Sign
{
    public string datetime;
    public string status;
    public BaseDoctor doctor;
    public Patient pacient;
    public Service service;
}

public class DB
{
    public string dbPath;

    public void delete(Sign sign) { }
    public void add(Sign sign) { }
}

public class Controller
{
    DB db;

    public bool createSign(
        Patient pacient,
        BaseDoctor doctor,
        Service service,
        string datetime
    )
    {
        if (!doctor.checkLicense()) return false;
        if (!doctor.checkService(service)) return false;
        if (!checkTime(doctor, datetime)) return false;
        Sign sign = new Sign();
        db.add(sign);
        return true;
    }
    public bool editSignDatetime(
        Sign sign,
        string datetime
    )
    { }
    public bool checkTime(
        BaseDoctor doctor,
        string datetime
    )
    {
        var schedule = doctor.getSchedule();
        //return !schedule.Contains(datetime);
        var dtAvailable = false;
        foreach (var dt in schedule)
        {
            if (dt == datetime)
            {
                dtAvailable = true;
                break;
            }
        }
        return dtAvailable;
    }
}
