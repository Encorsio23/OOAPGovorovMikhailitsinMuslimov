}

interface IControler
{
    public bool createSign(
        Patient pacient,
        Doctor doctor,
        Service service
    );
}

class Controller : IControler
{
    DB db;

    public bool createSign(
        Patient pacient,
        Doctor doctor,
        Service service,
        string datetime
    )
    {
        if (!doctor.checkLicense()) return false;
        if (!doctor.checkService(service)) return false;
        if (!checkTime(doctor, datetime)) return false;
        Sign sign = new Sign();
        db.add(sign);
        return true;
    }
    public bool editSignDatetime(
        Sign sign,
        string datetime
    )
    { }
    public bool checkTime(
        Doctor doctor,
        string datetime
    )
    {
        var schedule = doctor.getSchedule();
        //return !schedule.Contains(datetime);
        var dtAvailable = false;
        foreach (var dt in schedule)
        {
            if (dt == datetime)
            {
                dtAvailable = true;
                break;
            }
        }
        return dtAvailable;
    }
}
